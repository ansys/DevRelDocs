## Fatigue Analysis
This example demonstrates Fatigue Analysis.
**The example assumes that the material has already been defined.** When running it for the first time, create the material in advance.

Running this example runs a fatigue analysis on FEBody_01 with the S-N analysis type and names the fatigue result FEBody_01_Fatigue.
It then creates a contour of the Fatigue result with the Life Cycle component on FEBody_01.

This example uses the Tutorial_ModalFlex.dfr result file. You can get the Tutorial_ModalFlex.dfr file from [Introduction - Downloads](../../../index.md#downloads).
```python
# Example_OperationAPI_DurabilityAnalysis.py
import sys

# OPERATION_API_MODULE_PATH : path to the folder containing OperationAPI.py
sys.path.append(OPERATION_API_MODULE_PATH)

# Import necessary modules
from OperationAPI import *

# Start the headless application interface
application_handler = ApplicationHandler()

# Create a list of result file paths
filepaths = List[str]()
# RESULT_FILE_PATH : .dfr result file path (e.g., D:\PostprocessorAPI\Results\Tutorial_ModalFlex.dfr)
filepaths.Add(RESULT_FILE_PATH)

# Open the result file
# This opens the result file in the application.
# When a result file is opened for the first time, a page and an Animation View are created automatically.
application_handler.AddDocument(filepaths)

# AddDocument created a page with an animation view for the dynamics result.
page = application_handler.GetActivePage()
# The active view of the page.
active_view = page.GetActiveView()

# Load the animation data before running fatigue analysis.
application_handler.LoadingAnimation()

# Get instance of entity
# The GetViewModelByName method retrieves the target by its name.
# The name of the target finite-element body.
entity_name = "FEBody_01"
# The finite-element body with the specified name.
fe_body = active_view.GetViewModelByName(entity_name)
# The fatigue material must already be registered in the Postprocessor UI.
# The name of the fatigue material.
material_name = "2014_HV_O"
# The fatigue material of the body.
material = active_view.GetViewModelByName(material_name)

# Set FE body properties
fe_body.Material = material
fe_body.AnalysisType = FatigueAnalysisType.SN
fe_body.Scale = 2.0
fe_body.StressStrainCombination = StressStrainCombinationType.VonMises
fe_body.MeanStressCorrection = MeanStressCorrection.Neglect

# Set up fatigue analysis parameters
durability_param = DurabilityAnalysisParameter()
durability_param.DocumentFilePath = RESULT_FILE_PATH
# FATIGUE_RESULT_NAME : name of the fatigue result file in the Fatigue folder, without the .dffrf extension (e.g., FEBody_01_Fatigue)
durability_param.ResultName = FATIGUE_RESULT_NAME
durability_param.NoOfRepeatedLoad = 1
durability_param.Start = 1
durability_param.End = 10
durability_param.AddTarget(fe_body.ID)

# Run fatigue analysis
DurabilityAnalysis.RunFatigueAnalysis(durability_param)

# Create fatigue contours
# Fixed characteristic name required by the fatigue contour API.
fatigue_characteristic_name = "Fatigue"
# Fixed component name required for the fatigue life-cycle contour.
fatigue_life_cycle_component_name = "Life Cycle"
# The document opened from the result file.
document = application_handler.GetDocument(RESULT_FILE_PATH)
# The analysis result of the document.
analysis_result = document.GetAnalysisResultViewModel(AnalysisResultType.Dynamics)
# The names of the target entities.
target_entities = List[str]()
target_entities.Add(fe_body.FullName)
analysis_result.CreateContour(target_entities, ContourMappingType.FEElement, fatigue_characteristic_name, fatigue_life_cycle_component_name)
```

## Sound Pressure
Sound pressure analysis (SPA) can be used to analyze the sound pressure generated by the vibration of the system surface based on "Rayleigh Integral". This example describes how to create the Sound Pressure.
* Before the analysis, the bodies that are the source of sound pressure must be defined as FE or EasyFlex bodies in the preprocessor.
* The coordinate systems must be created to define the microphone positions which are the target positions for sound pressure calculation.

Running this example creates a coordinate system named Microphone on Ground with a position offset of 300 along the Y axis.
It then creates sound pressure curve data for FEBody_01 at the Microphone position with a frequency of 2000.

This example uses the Tutorial_ModalFlex.dfr result file. You can get the Tutorial_ModalFlex.dfr file from [Introduction - Downloads](../../../index.md#downloads).
```python
# Example_OperationAPI_CreatingASoundPressure.py
import sys

# OPERATION_API_MODULE_PATH : path to the folder containing OperationAPI.py
sys.path.append(OPERATION_API_MODULE_PATH)

# Import necessary modules
from OperationAPI import *

# Start the headless application interface
application_handler = ApplicationHandler()

# Create a list of result file paths
filepaths = List[str]()
# RESULT_FILE_PATH : .dfr result file path (e.g., D:\PostprocessorAPI\Results\Tutorial_ModalFlex.dfr)
filepaths.Add(RESULT_FILE_PATH)

# Open the result file
# This opens the result file in the application.
# When a result file is opened for the first time, a page and an Animation View are created automatically.
application_handler.AddDocument(filepaths)

# Get all created pages.
# This retrieves all pages created in the application.
pages = application_handler.GetPages()

# The animation views that show the dynamic analysis result of the result file.
find_views = list()
for page in pages :
    # The views of the page.
    views = page.GetViews()
    # The animation views of the page.
    animation_views = [view for view in views if view.ViewType == ViewType.Animation]
    for animation_view in animation_views :
        if animation_view.DocumentFilePath == RESULT_FILE_PATH and animation_view.AnalysisResultType == AnalysisResultType.Dynamics :
            find_views.append(animation_view)

# The number of animation views found.
view_count = len(find_views)
if view_count > 0 :
    animation_view = find_views[0]

    # Create a Coordinate System
    # Name of the new coordinate system used as the microphone position.
    microphone_name = "Microphone"
    # Full name of the parent entity that contains the coordinate system.
    parent_name = "Ground"
    # The coordinate system created as the microphone position.
    microphone = animation_view.CreateCoordinateSystem(microphone_name, parent_name)
    microphone.TransformationOffsetPosition = Vector(0,300,0)

    # Full name of the entity whose surface generates the sound pressure.
    body_name = r'FEBody_01'
    # The full names of the entities whose surfaces generate the sound pressure.
    body_names = List[str](1)
    body_names.Add(body_name)
    # The names of the coordinate systems used as microphone positions.
    microphone_names = List[str](1)
    microphone_names.Add(microphone_name)

    # Creates sound pressure data based on the Rayleigh Integral for analyzing sound pressure generated by vibrations on the system surface.
    # Percentage of the number of nodes remaining in the sampling result with respect to the initial number of nodes.
    coarsening_rate = 10.0
    # Maximum allowable area of each element generated by the sampling result, so that the area of each element cannot be greater than (Max Area Factor) * (Average element area).
    max_area_factor = 100.0
    # Maximum allowable aspect ratio of each element generated by the sampling result.
    max_aspect_ratio = 3.0
    # Density of the air.
    air_density = 1.21E-09
    # Speed of sound in air.
    sound_velocity = 343000.0
    # Analysis frequency, which determines the number of analysis results per second as 10*analysis frequency.
    frequency = 2000.0
    # Start frame of the time range to be analyzed.
    start = 1
    # End frame of the time range to be analyzed.
    end = 10
    # True to return decibel-scaled results.
    is_decibel_scale = False
    animation_view.CreateSoundPressure(body_names, microphone_names, coarsening_rate, max_area_factor, max_aspect_ratio, air_density, sound_velocity, frequency, start, end, is_decibel_scale)
```
